// app.js // Application de discussion complète — TOUT est ici, dans un seul fichier : // - base de données SQLite (créée automatiquement) // - API REST (inscription / connexion / historique) // - WebSocket temps réel (Socket.IO) // - page HTML/CSS/JS servie directement par le serveur // // Lancement : // npm install // npm start // Puis ouvrez http://localhost:3000 dans plusieurs onglets. const express = require('express'); const http = require('http'); const bcrypt = require('bcryptjs'); const jwt = require('jsonwebtoken'); const { Server } = require('socket.io'); const Database = require('better-sqlite3'); // --------------------------------------------------------------------------- // Base de données // --------------------------------------------------------------------------- const db = new Database('chat.db'); db.pragma('journal_mode = WAL'); db.exec(` CREATE TABLE IF NOT EXISTS users ( id INTEGER PRIMARY KEY AUTOINCREMENT, phone TEXT UNIQUE NOT NULL, password_hash TEXT NOT NULL, created_at TEXT NOT NULL DEFAULT (datetime('now')) ); CREATE TABLE IF NOT EXISTS messages ( id INTEGER PRIMARY KEY AUTOINCREMENT, username TEXT NOT NULL, text TEXT NOT NULL, created_at TEXT NOT NULL ); `); // --------------------------------------------------------------------------- // Serveur // --------------------------------------------------------------------------- const app = express(); const server = http.createServer(app); const io = new Server(server); const JWT_SECRET = process.env.JWT_SECRET || 'change-this-secret-in-production'; if (!process.env.JWT_SECRET) { console.warn('⚠️ JWT_SECRET non défini : une clé par défaut est utilisée (à ne jamais faire en production).'); } app.use(express.json()); // ---- Inscription ---- app.post('/api/register', (req, res) => { const { phone, password } = req.body || {}; if (!phone || !password) { return res.status(400).json({ error: 'Numéro et mot de passe requis.' }); } if (password.length < 4) { return res.status(400).json({ error: 'Mot de passe trop court (4 caractères minimum).' }); } const existing = db.prepare('SELECT id FROM users WHERE phone = ?').get(phone); if (existing) { return res.status(409).json({ error: 'Ce numéro est déjà enregistré.' }); } const hash = bcrypt.hashSync(password, 10); const info = db.prepare('INSERT INTO users (phone, password_hash) VALUES (?, ?)').run(phone, hash); const token = jwt.sign({ id: info.lastInsertRowid, phone }, JWT_SECRET, { expiresIn: '7d' }); res.json({ token, phone }); }); // ---- Connexion ---- app.post('/api/login', (req, res) => { const { phone, password } = req.body || {}; if (!phone || !password) { return res.status(400).json({ error: 'Numéro et mot de passe requis.' }); } const user = db.prepare('SELECT * FROM users WHERE phone = ?').get(phone); if (!user || !bcrypt.compareSync(password, user.password_hash)) { return res.status(401).json({ error: 'Numéro ou mot de passe incorrect.' }); } const token = jwt.sign({ id: user.id, phone: user.phone }, JWT_SECRET, { expiresIn: '7d' }); res.json({ token, phone: user.phone }); }); // ---- Historique des messages ---- function authMiddleware(req, res, next) { const header = req.headers.authorization || ''; const token = header.replace('Bearer ', ''); if (!token) return res.status(401).json({ error: 'Non authentifié.' }); try { req.user = jwt.verify(token, JWT_SECRET); next(); } catch { res.status(401).json({ error: 'Session expirée, reconnectez-vous.' }); } } app.get('/api/messages', authMiddleware, (req, res) => { const rows = db.prepare('SELECT username, text, created_at FROM messages ORDER BY id DESC LIMIT 50').all(); res.json(rows.reverse()); }); // ---- Page unique (HTML/CSS/JS servis directement, pas de dossier public) ---- app.get('/', (req, res) => { res.type('html').send(PAGE_HTML); }); // --------------------------------------------------------------------------- // WebSocket temps réel // --------------------------------------------------------------------------- const onlineUsers = new Map(); // socket.id -> phone io.use((socket, next) => { try { const token = socket.handshake.auth && socket.handshake.auth.token; socket.user = jwt.verify(token, JWT_SECRET); next(); } catch { next(new Error('Authentification requise.')); } }); io.on('connection', (socket) => { onlineUsers.set(socket.id, socket.user.phone); io.emit('presence', Array.from(new Set(onlineUsers.values()))); socket.on('chat:message', (text) => { const clean = String(text || '').trim().slice(0, 2000); if (!clean) return; const created_at = new Date().toISOString(); db.prepare('INSERT INTO messages (username, text, created_at) VALUES (?, ?, ?)') .run(socket.user.phone, clean, created_at); io.emit('chat:message', { username: socket.user.phone, text: clean, created_at }); }); socket.on('disconnect', () => { onlineUsers.delete(socket.id); io.emit('presence', Array.from(new Set(onlineUsers.values()))); }); }); const PORT = process.env.PORT || 3000; server.listen(PORT, () => { console.log(`Serveur lancé sur http://localhost:${PORT}`); }); // --------------------------------------------------------------------------- // Frontend (HTML + CSS + JS) — une seule chaîne de caractères, aucun fichier // externe à part socket.io.js qui est déjà servi automatiquement par le // serveur Socket.IO sur /socket.io/socket.io.js // --------------------------------------------------------------------------- const PAGE_HTML = ` Application de Discussion

Connexion

Pas encore de compte ? Créer un compte
En ligne
Salon Public
`;
